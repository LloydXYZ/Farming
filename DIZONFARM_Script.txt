function toast(msg,type="success"){
 const host=document.getElementById("toast");
 if(!host){alert(msg);return;}
 // Each popup gets its own element so multiple notifications do not overwrite one another.
 const item=document.createElement("div");
 item.className="toast-message "+(type==="error"?"error":"success");
 item.setAttribute("role",type==="error"?"alert":"status");
 item.setAttribute("aria-live",type==="error"?"assertive":"polite");
 item.textContent=msg;
 host.appendChild(item);
 requestAnimationFrame(()=>item.classList.add("show"));
 const timer=setTimeout(()=>{
   item.classList.remove("show");
   setTimeout(()=>item.remove(),250);
 },5000);
 item.addEventListener("click",()=>{clearTimeout(timer);item.remove();});
}

function calendar(eid,selected){const now=new Date(),y=now.getFullYear(),m=now.getMonth(),first=new Date(y,m,1).getDay(),days=new Date(y,m+1,0).getDate(),names=["Sun","Mon","Tue","Wed","Thu","Fri","Sat"];let h='<div class="calendar">'+names.map(x=>`<div class="cal-head">${x}</div>`).join("");for(let i=0;i<first;i++)h+="<div></div>";for(let day=1;day<=days;day++){let dt=`${y}-${String(m+1).padStart(2,"0")}-${String(day).padStart(2,"0")}`,isB=availableUnitsOnDate(eid,dt)<=0,isS=selected===dt;h+=`<div class="cal-day ${isB?"booked":"available"} ${isS?"selected":""}">${day}${isB?"<br><small>Fully booked</small>":""}</div>`}return h+"</div>"}
function nextBooking(){
 try{
  if(!selectedEquipment){toast("Please select equipment first.","error");return false}
  const ids=["bDate","bTime","bHa","bPurok","bBarangay","bMunicipality"],v={};
  ids.forEach(id=>{const el=document.getElementById(id);v[id]=el?String(el.value||"").trim():""});
  if(ids.some(id=>!v[id])){toast("Please complete all required booking fields.","error");return false}
  if(Number(v.bHa)<=0){toast("Hectare area must be greater than zero.","error");return false}
  const chosen=new Date(v.bDate+"T"+v.bTime);
  if(Number.isNaN(chosen.getTime())){toast("Please enter a valid date and time.","error");return false}
  if(chosen.getTime()<Date.now()){toast("Please choose a future date and time.","error");return false}
  if(availableUnits(selectedEquipment.id)<=0){toast("This equipment is no longer available.","error");return false}
  if(availableUnitsOnDate(selectedEquipment.id,v.bDate)<=0){toast("That date is fully reserved. Please select another date.","error");return false}
  bookingDraft={...bookingDraft,equipmentId:selectedEquipment.id,date:v.bDate,time:v.bTime,ha:v.bHa,purok:v.bPurok,barangay:v.bBarangay,municipality:v.bMunicipality};
  bookingStep=2;
  renderBookingModal();
  toast("Step 1 complete. Review your booking and choose a payment method.","success");
  return true;
 }catch(err){console.error(err);toast("Unable to continue. Please check the booking details and try again.","error");return false}
}
function nextPayment(){
 try{
  const p=document.querySelector('input[name="pay"]:checked');
  if(!p){toast("Please select a payment method.","error");return false}
  bookingDraft.payment=p.value;
  bookingStep=3;
  renderBookingModal();
  toast("Payment method saved. Review the confirmation details.","success");
  return true;
 }catch(err){console.error(err);toast("Unable to continue to confirmation.","error");return false}
}
function confirmBooking(){
 try{
  const u=currentUser(),e=selectedEquipment;
  if(!u){toast("Please log in as a farmer.","error");return false}
  if(!e){toast("Equipment selection is missing.","error");return false}
  if(!bookingDraft.date||!bookingDraft.time||!bookingDraft.payment){toast("Please complete all booking steps first.","error");bookingStep=1;renderBookingModal();return false}
  if(availableUnits(e.id)<=0||availableUnitsOnDate(e.id,bookingDraft.date)<=0){toast("This equipment is no longer available for the selected date.","error");bookingStep=1;renderBookingModal();return false}
  const now=Date.now(),code="BK-"+String(now).slice(-6),b={id:now,code,userId:u.id,equipmentId:e.id,date:bookingDraft.date,time:bookingDraft.time,ha:Number(bookingDraft.ha),purok:bookingDraft.purok,barangay:bookingDraft.barangay,municipality:bookingDraft.municipality,total:e.rate,payment:bookingDraft.payment,status:"Pending",createdAt:new Date().toISOString()};
  db.bookings.push(b);
  const paid=b.payment!=="Cash";
  db.payments.push({id:now+1,bookingId:b.id,userId:u.id,date:new Date().toISOString().slice(0,10),amount:b.total,method:b.payment,status:paid?"Paid":"Pending",reference:"PAY-"+String(now).slice(-6)});
  notifyAndToast(u.id,"Order Confirmation",`Your booking ${b.code} for ${e.name} on ${b.date} at ${b.time} was filed successfully.`,"order");
  notifyAndToast(u.id,"Payment Confirmation",paid?`Payment of ${money(b.total)} via ${b.payment} for ${b.code} was recorded.`:`Your ${b.payment} payment method for ${b.code} was recorded and is pending confirmation.`,"payment");
  save();
  closeModal();
  toast(`Booking ${b.code} filed successfully! Availability updated.`,"success");
  navigate("bookings");
  return true;
 }catch(err){console.error(err);toast("Booking could not be filed. Please try again.","error");return false}
}

function deleteEquipment(id){
 if(session?.role!=="admin"){toast("Only administrators can delete equipment.","error");return}
 const equipment=db.equipment.find(e=>e.id===id);if(!equipment){toast("Equipment could not be found.","error");return}
 const active=db.bookings.filter(b=>b.equipmentId===id&&["Pending","Confirmed"].includes(b.status));
 if(active.length){toast("Cannot delete this equipment while it has pending or confirmed bookings. Cancel or complete those bookings first.","error");return}
 if(!confirm(`Delete ${equipment.name} (${equipment.unit})? This action cannot be undone.`))return;
 db.equipment=db.equipment.filter(e=>e.id!==id);save();toast("Equipment deleted successfully.");render();
}
